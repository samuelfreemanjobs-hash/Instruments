cmake_minimum_required(VERSION 3.22)
project(MpcPlayTemplate VERSION 1.0.0)

option(MPC_VST3_HEADLESS "Melodee headless build" ON)

if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../JUCE/CMakeLists.txt")
  message(FATAL_ERROR "JUCE not found. git submodule update --init vst/JUCE")
endif()
add_subdirectory(../JUCE JUCE)

juce_add_plugin(MpcPlayTemplate
    COMPANY_NAME "Instruments"
    BUNDLE_ID com.instruments.mpcplaytemplate
    PLUGIN_MANUFACTURER_CODE Inst
    PLUGIN_CODE Mptm
    FORMATS VST3 Standalone
    PRODUCT_NAME "MpcPlayTemplate"
    IS_SYNTH TRUE
    NEEDS_MIDI_INPUT TRUE
    NEEDS_MIDI_OUTPUT FALSE
    IS_MIDI_EFFECT FALSE
    EDITOR_WANTS_KEYBOARD_FOCUS FALSE
    COPY_PLUGIN_AFTER_BUILD FALSE)

set(PLAY_CORE "${CMAKE_CURRENT_SOURCE_DIR}/../../templates/juce8-synth-base/Source")

target_sources(MpcPlayTemplate
    PRIVATE
        Source/PluginProcessor.cpp
        Source/FactoryPrograms.h
        ${PLAY_CORE}/PlaySynthVoice.cpp
        ${PLAY_CORE}/PerformanceEngine.cpp)

target_compile_definitions(MpcPlayTemplate
    PUBLIC
        JUCE_WEB_BROWSER=0
        JUCE_USE_CURL=0
        JUCE_VST3_CAN_REPLACE_VST2=0)

target_link_libraries(MpcPlayTemplate
    PRIVATE
        juce::juce_audio_utils
        juce::juce_dsp
    PUBLIC
        juce::juce_recommended_config_flags
        juce::juce_recommended_warning_flags)

target_include_directories(MpcPlayTemplate PRIVATE Source "${PLAY_CORE}")
juce_generate_juce_header(MpcPlayTemplate)
